Run: pip install yt-dlp", ) except subprocess.TimeoutExpired: raise HTTPException(status_code=504, detail="Download timed out") if proc.returncode != 0 or not out_file.exists(): err = (proc.stderr or proc.stdout or "yt-dlp failed").strip()[-300:] raise HTTPException(status_code=502, detail=err) return FileResponse( path=str(out_file), media_type="video/mp4", filename=f"vantyx_{key}.mp4", ) if __name__ == "__main__": import uvicorn port = int(os.environ.get("PORT", "8000")) uvicorn.run(app, host="0.0.0.0", port=port)
